# VOID — шпаргалка (1 страница)

```
# комментарий
шепот "тоже комментарий"

# переменные
связать x = 42;           bind x = 42;
связать имя = "VOID";
пустота                   void / None

# ленивые вычисления
парадокс p => 10 / 0 ?! 42;
коллапс p -> y;           # y = 42

# запасная реальность
a = пустота ?! "план Б";  # если слева невозможно — взять справа
a = (10/0) ?! 100;        # 100
a = пустота ?! пустота ?! "наконец";

# вывод
эхо "привет " + имя;      echo / print

# условия
если x > 5 { эхо "больше"; } иначе { эхо "меньше"; }
if x > 5 { echo "big"; } else { echo "small"; }

# циклы
пока i < 10 { i = i + 1; }    while i < 10 { ... }
петля { если k==5 { остановить; } }   loop { if k==5 { break; } }
для i в 0..10 { эхо i; }      for i in 0..10 { echo i; }
для v в [1,2,3] { эхо v; }

# функции
призвать фиб(n) {
    если n <= 1 { вернуть n; }
    вернуть фиб(n-1) + фиб(n-2);
}
summon fib(n) { if n <= 1 { return n; } return fib(n-1)+fib(n-2); }

# массивы и диапазоны
[1,2,3]                    # массив
0..10                      # диапазон 0..9
длина([1,2,3])             len(arr) = 3

# операторы
+ - * / %                  # арифметика
== != < > <= >=           # сравнение
и / && / and              # И
или / || / or             # ИЛИ
! ~ не                    # НЕ
-> => ?! ..               # спец VOID

# встроенные
длина(x) / len(x)
строка(x) / str(x)
число(s) / num(s)
тип(x) / type(x)
время() / time()
случайно() / random()

# типы (динамические)
число 3.14, строка "hi", булево истина/ложь, пустота, массив, функция, парадокс

# запуск
python void.py file.void      # выполнить
python void.py --repl         # REPL
python void.py --ast file     # AST
python void.py --tokens file  # токены
```
